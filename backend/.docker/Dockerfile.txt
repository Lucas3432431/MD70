# ===============================================
# Backend - Development Hot Reload v1.0
# ===============================================
# Python FastAPI + Playwright + Hot reload

# ===============================================
# Stage 1: Builder (cache dependencies)
# ===============================================
FROM docker.io/library/python:3.11-slim AS builder

WORKDIR /app

# Install build dependencies
RUN echo "[INFO] Installing build dependencies" && \
    rm -f /var/lib/apt/lists/lock /var/cache/apt/archives/lock /var/lib/dpkg/lock* && \
    apt-get update && apt-get install -y --no-install-recommends \
        build-essential \
        libffi-dev \
        libssl-dev \
        curl \
        ca-certificates \
        git \
        && echo "[OK] Build dependencies installed" && \
    rm -rf /var/lib/apt/lists/*

# Copy requirements
COPY requirements.txt ./

# Install Python dependencies in a venv
RUN echo "[INFO] Installing Python dependencies" && \
    python -m venv /opt/venv && \
    /opt/venv/bin/pip install --upgrade pip && \
    /opt/venv/bin/pip install -r requirements.txt && \
    echo "[OK] Python dependencies installed"

# ===============================================
# Stage 2: Runtime (final image)
# ===============================================
FROM docker.io/library/python:3.11-slim AS runtime

ARG BACKEND_PORT=4000
ARG NODE_ENV=development
ARG HOT_RELOAD_ENABLED=true

ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1 \
    PIP_NO_CACHE_DIR=1 \
    PIP_DISABLE_PIP_VERSION_CHECK=1 \
    NODE_ENV=${NODE_ENV} \
    BACKEND_PORT=${BACKEND_PORT} \
    HOT_RELOAD_ENABLED=${HOT_RELOAD_ENABLED} \
    PYTHONPATH=/app:/app/src \
    PATH="/opt/venv/bin:$PATH"

WORKDIR /app

# Install runtime dependencies only
RUN echo "[INFO] Installing runtime dependencies" && \
    rm -f /var/lib/apt/lists/lock /var/cache/apt/archives/lock /var/lib/dpkg/lock* && \
    apt-get update && apt-get install -y --no-install-recommends \
        curl \
        ca-certificates \
        libffi8 \
        libssl3 \
        bash \
        procps \
        inotify-tools \
        && echo "[OK] Runtime dependencies installed" && \
    rm -rf /var/lib/apt/lists/*

# Create app user
RUN addgroup --gid 1000 appuser && \
    adduser --uid 1000 --gid 1000 --disabled-password appuser && \
    mkdir -p /app/src /app/scripts /app/logs /app/data && \
    chown -R appuser:appuser /app

# Copy venv from builder
COPY --from=builder /opt/venv /opt/venv

# Copy application code
COPY --chown=appuser:appuser requirements.txt ./
COPY --chown=appuser:appuser . .

# Install Playwright browsers
RUN echo "[INFO] Installing Playwright browsers" && \
    /opt/venv/bin/playwright install && \
    echo "[OK] Playwright browsers installed"

# Create entrypoint script
RUN echo '#!/bin/bash' > /app-entrypoint.sh && \
    echo 'set -e' >> /app-entrypoint.sh && \
    echo 'UUID=$(openssl rand -hex 4 2>/dev/null || echo "0000")' >> /app-entrypoint.sh && \
    echo 'log_info() { echo "[$UUID] [INFO] [Backend.Entrypoint] $1"; }' >> /app-entrypoint.sh && \
    echo 'log_ok() { echo "[$UUID] [OK] [Backend.Entrypoint] $1"; }' >> /app-entrypoint.sh && \
    echo 'log_debug() { echo "[$UUID] [DEBUG] [Backend.Entrypoint] $1"; }' >> /app-entrypoint.sh && \
    echo '' >> /app-entrypoint.sh && \
    echo 'log_info "Starting Backend - FastAPI Hot Reload v1.0"' >> /app-entrypoint.sh && \
    echo 'log_debug "Python Watchdog + FastAPI + Playwright"' >> /app-entrypoint.sh && \
    echo 'log_debug "Backend Port: ${BACKEND_PORT}"' >> /app-entrypoint.sh && \
    echo 'log_debug "Environment: ${NODE_ENV}"' >> /app-entrypoint.sh && \
    echo '' >> /app-entrypoint.sh && \
    echo 'if [ "$HOT_RELOAD_ENABLED" = "true" ] && [ "$NODE_ENV" = "development" ]; then' >> /app-entrypoint.sh && \
    echo '    log_info "Python Watchdog hot reload enabled"' >> /app-entrypoint.sh && \
    echo '    log_debug "Validating dependencies..."' >> /app-entrypoint.sh && \
    echo '    python -c "import fastapi" && log_ok "FastAPI available"' >> /app-entrypoint.sh && \
    echo '    python -c "import playwright" && log_ok "Playwright available"' >> /app-entrypoint.sh && \
    echo '    python -c "from watchdog.observers import Observer" && log_ok "Watchdog available"' >> /app-entrypoint.sh && \
    echo '    log_info "Starting with hot reload..."' >> /app-entrypoint.sh && \
    echo '    exec python -m watchdog.watchmedo auto-restart \' >> /app-entrypoint.sh && \
    echo '        --directory=/app \' >> /app-entrypoint.sh && \
    echo '        --pattern="*.py;*.json;*.txt;*.env;requirements.txt" \' >> /app-entrypoint.sh && \
    echo '        --recursive \' >> /app-entrypoint.sh && \
    echo '        --ignore-patterns="*data/*;*logs/*;*.db;*.sqlite*;*.log;*.tmp;*.pyc;*__pycache__*;*.zip;*.backup;*storage*" \' >> /app-entrypoint.sh && \
    echo '        --debounce-interval=2 \' >> /app-entrypoint.sh && \
    echo '        -- python "$@"' >> /app-entrypoint.sh && \
    echo 'else' >> /app-entrypoint.sh && \
    echo '    log_info "Production mode"' >> /app-entrypoint.sh && \
    echo '    exec python "$@"' >> /app-entrypoint.sh && \
    echo 'fi' >> /app-entrypoint.sh && \
    chmod +x /app-entrypoint.sh

LABEL com.mvp-zera.service.name="backend" \
      com.mvp-zera.service.type="development-fastapi-watchdog" \
      com.mvp-zera.framework="fastapi-python" \
      com.mvp-zera.tools="playwright,crawl4ai,anthropic,openai" \
      com.mvp-zera.hot-reload.enabled="true" \
      com.mvp-zera.version="1.0-development"

USER appuser

EXPOSE ${BACKEND_PORT}

ENTRYPOINT ["/app-entrypoint.sh"]
HEALTHCHECK --interval=30s --timeout=10s --start-period=60s --retries=3 \
    CMD python3 -c "import socket; s=socket.socket(); s.settimeout(3); s.connect(('127.0.0.1', 4001)); s.close()"

CMD ["Main.py"]
